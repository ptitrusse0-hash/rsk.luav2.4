local SAVE_DIR="RSK_Configs/"; local SAVE_EXT=".cfg"
local function encodeProfile(name)
    local lines={"[RSK_PROFILE:"..name.."]"}
    for _,k in ipairs(PROFILE_KEYS) do
        local v=CONFIG[k]; local t=typeof(v)
        if t=="boolean" then table.insert(lines,k.."=bool:"..tostring(v))
        elseif t=="number" then table.insert(lines,k.."=num:"..tostring(v))
        elseif t=="string" then table.insert(lines,k.."=str:"..v) end
    end
    return table.concat(lines,"\n")
end
local function decodeProfile(raw)
    local data={}
    for line in raw:gmatch("[^\n]+") do
        local k,typ,val=line:match("^(.-)=(%a+):(.+)$")
        if k and typ and val then
            if typ=="bool" then data[k]=(val=="true")
            elseif typ=="num" then data[k]=tonumber(val) or 0
            elseif typ=="str" then data[k]=val end
        end
    end
    return data
end
local function applyProfile(data)
    for _,k in ipairs(PROFILE_KEYS) do if data[k]~=nil then CONFIG[k]=data[k] end end
end
local function saveProfile(name)
    if not writefile then return false,"writefile non supporté" end
    local ok,err=pcall(function()
        if not isfolder(SAVE_DIR) then makefolder(SAVE_DIR) end
        writefile(SAVE_DIR..name..SAVE_EXT,encodeProfile(name))
    end)
    return ok, ok and "sauvegardé" or tostring(err)
end
local function loadProfile(name)
    if not readfile then return false,"readfile non supporté" end
    local ok,raw=pcall(readfile,SAVE_DIR..name..SAVE_EXT)
    if not ok then return false,"fichier introuvable" end
    applyProfile(decodeProfile(raw)); return true,"chargé"
end
local function listSavedProfiles()
    if not listfiles then return {} end
    local ok,files=pcall(listfiles,SAVE_DIR); if not ok then return {} end
    local names={}
    for _,path in ipairs(files) do
        local n=path:match("([^/\\]+)"..SAVE_EXT.."$")
        if n then table.insert(names,n) end
    end
    return names
end
local function deleteProfile(name)
    if not delfile then return false,"delfile non supporté" end
    local ok,err=pcall(delfile,SAVE_DIR..name..SAVE_EXT)
    return ok, ok and "supprimé" or tostring(err)
end

local function worldToViewport(pos)
    local vp,onScreen=Camera:WorldToViewportPoint(pos); return Vector2.new(vp.X,vp.Y),onScreen,vp.Z
end
local function getPart(char,name) return char and char:FindFirstChild(name) end
local function getHumanoid(char) return char and char:FindFirstChildOfClass("Humanoid") end
local function isAlive(p) local c=p.Character; local h=getHumanoid(c); return c and h and h.Health>0 end
local function getEnemies()
    local t={}
    for _,p in ipairs(Players:GetPlayers()) do
        if p~=LocalPlayer and isAlive(p) then table.insert(t,p) end
    end
    return t
end
local function isVisible(part)
    if not CONFIG.VisibleOnly then return true end
    local char=LocalPlayer.Character; local origin=char and char:FindFirstChild("HumanoidRootPart")
    if not origin then return true end
    local params=RaycastParams.new(); params.FilterType=Enum.RaycastFilterType.Exclude
    local ignore={}
    for _,p in ipairs(Players:GetPlayers()) do if p.Character then table.insert(ignore,p.Character) end end
    params.FilterDescendantsInstances=ignore
    return workspace:Raycast(origin.Position,part.Position-origin.Position,params)==nil
end
local function getDistance(player)
    local mc=LocalPlayer.Character; local hr=mc and mc:FindFirstChild("HumanoidRootPart")
    local ec=player.Character; local er=ec and ec:FindFirstChild("HumanoidRootPart")
    if not hr or not er then return 0 end
    return math.floor((hr.Position-er.Position).Magnitude)
end
local function getHealthPercent(player)
    local c=player.Character; local h=getHumanoid(c)
    if not h then return 0 end; return math.clamp(h.Health/h.MaxHealth,0,1)
end
local function lerpColor(a,b,t)
    return Color3.new(a.R+(b.R-a.R)*t,a.G+(b.G-a.G)*t,a.B+(b.B-a.B)*t)
end
local function getFrenchTime()
    local utcH=math.floor(os.time()/3600)%24; local frH=(utcH+1)%24
    local m=math.floor(os.time()/60)%60; local s=os.time()%60
    return string.format("%02d:%02d:%02d",frH,m,s)
end
local function getBestTarget()
    local myChar=LocalPlayer.Character; if not myChar then return nil,nil end
    local center=Vector2.new(Camera.ViewportSize.X/2,Camera.ViewportSize.Y/2)
    local best,bestPart,bestDist=nil,nil,CONFIG.AimFOV
    for _,enemy in ipairs(getEnemies()) do
        local char=enemy.Character
        if char then
            local part=getPart(char,CONFIG.AimPart) or getPart(char,"HumanoidRootPart")
            if part then
                local sp,onScreen=worldToViewport(part.Position)
                if onScreen then
                    local dist=(sp-center).Magnitude
                    if dist<bestDist then
                        if not CONFIG.VisibleOnly or isVisible(part) then
                            bestDist=dist; best=enemy; bestPart=part
                        end
                    end
                end
            end
        end
    end
    return bestPart,best
end
local lastRemoteFire=0
local function fireRemote(remoteName,...)
    local now=tick()
    if (now-lastRemoteFire)<0.12 then task.wait(0.12-(now-lastRemoteFire)) end
    lastRemoteFire=tick()
    local args={...}; local rs=game:GetService("ReplicatedStorage")
    local remote=rs:FindFirstChild(remoteName,true)
    if remote and remote:IsA("RemoteEvent") then
        remote:FireServer(table.unpack(args)); return true,"OK"
    elseif remote and remote:IsA("RemoteFunction") then
        local ok,res=pcall(function() return remote:InvokeServer(table.unpack(args)) end)
        return ok,tostring(res)
    end
    return false,"remote introuvable: "..remoteName
end
local function getEquippedWeapon()
    local char=LocalPlayer.Character; if not char then return nil end
    for _,obj in ipairs(char:GetChildren()) do if obj:IsA("Tool") then return obj end end
    return nil
end
local function getWeaponName()
    local tool=getEquippedWeapon(); if not tool then return nil end; return tool.Name
end

if hookmetamethod then
    local oldNamecall
    oldNamecall=hookmetamethod(game,"__namecall",function(self,...)
        local method=getnamecallmethod()
        if CONFIG.DumpRemote and (method==sFireServer or method==sInvokeServer) then
            print(string.format("[RSK DUMP] %s | method=%s",self:GetFullName(),method))
            local args={...}
            for i,v in ipairs(args) do print(string.format("  arg%d  type=%-12s  val=%s",i,typeof(v),tostring(v))) end
            return oldNamecall(self,...)
        end
        if method==sFireServer and CONFIG.SilentAim and CONFIG.AimbotEnabled
            and aimActive and tostring(self.Name):find(CONFIG.RemoteNameFilter)
        then
            local target=getBestTarget()
            if target then
                local args=table.pack(...)
                local dir=(target.Position-Camera.CFrame.Position).Unit
                args[CONFIG.RemoteArgDir]=dir
                if CONFIG.RemoteArgPart<=args.n then args[CONFIG.RemoteArgPart]=target end
                return oldNamecall(self,table.unpack(args,1,args.n))
            end
        end
        return oldNamecall(self,...)
    end)
else
    warn("[RSK] hookmetamethod indisponible — silent aim serveur désactivé")
end

UserInputService.InputBegan:Connect(function(inp,gpe)
    if gpe then return end

    -- rebind stream key
    if streamKeyListening and inp.UserInputType==Enum.UserInputType.Keyboard then
        CONFIG.StreamKey=inp.KeyCode
        streamKeyBtn.Text=tostring(inp.KeyCode):gsub("Enum.KeyCode.","")
        streamKeyBtn.TextColor3=Color3.fromRGB(255,255,255); streamKeyListening=false; return
    end
    if panicKeyListening and inp.UserInputType==Enum.UserInputType.Keyboard then
        CONFIG.PanicKey=inp.KeyCode
        panicKeyBtn.Text=tostring(inp.KeyCode):gsub("Enum.KeyCode.","")
        panicKeyBtn.TextColor3=Color3.fromRGB(255,100,100); panicKeyListening=false; return
    end

    -- stream key toggle
    if inp.KeyCode==CONFIG.StreamKey then
        if streamActive then
            exitStreamMode()
            CONFIG.MenuOpen=true; MenuGui.Enabled=true
        else
            enterStreamMode()
            CONFIG.MenuOpen=false
        end
        refreshStreamBanner(); return
    end

    -- panic key hold
    if inp.KeyCode==CONFIG.PanicKey then
        panicHide(); CONFIG.MenuOpen=false; MenuGui.Enabled=false; return
    end

    -- menu toggle (bloqué en stream/panic)
    if inp.KeyCode==CONFIG.MenuKey then
        if streamActive or panicActive then return end
        CONFIG.MenuOpen=not CONFIG.MenuOpen; MenuGui.Enabled=CONFIG.MenuOpen
    end

    if inp.KeyCode==CONFIG.AimKey then
        if CONFIG.AimHoldMode then aimActive=true else aimActive=not aimActive end
    end
    if inp.UserInputType==Enum.UserInputType.MouseButton1 then mouseDown=true end
end)

UserInputService.InputEnded:Connect(function(inp)
    if inp.KeyCode==CONFIG.AimKey and CONFIG.AimHoldMode then aimActive=false end
    if inp.UserInputType==Enum.UserInputType.MouseButton1 then mouseDown=false end
    -- panic key relâché = restore
    if inp.KeyCode==CONFIG.PanicKey then panicRestore() end
end)

    for _,obj in ipairs(espData[player].hpBars or {}) do pcall(function() obj:Remove() end) end
    if espData[player].nametag then pcall(function() espData[player].nametag:Remove() end) end
    if espData[player].distLbl then pcall(function() espData[player].distLbl:Remove() end) end
    espData[player]=nil
end
local function initESP(player)
    if espData[player] then return end
    local lines={}
    for _=1,#SKELETON_CONNECTIONS do
        local l=Drawing.new("Line"); l.Visible=false; l.Color=CONFIG.SkeletonColor; l.Thickness=CONFIG.ESPThickness
        table.insert(lines,l)
    end
    local box={}
    for _=1,4 do local l=Drawing.new("Line"); l.Visible=false; l.Color=CONFIG.BoxColor; l.Thickness=1.2; table.insert(box,l) end
    local hpBack=Drawing.new("Line"); hpBack.Visible=false; hpBack.Color=Color3.fromRGB(30,30,30); hpBack.Thickness=4
    local hpFill=Drawing.new("Line"); hpFill.Visible=false; hpFill.Color=Color3.fromRGB(80,255,120); hpFill.Thickness=3
    local nametag=Drawing.new("Text"); nametag.Visible=false; nametag.Color=CONFIG.NametagColor
    nametag.Size=13; nametag.Outline=true; nametag.Center=true; nametag.Font=Drawing.Fonts.UI
    local distLbl=Drawing.new("Text"); distLbl.Visible=false; distLbl.Color=Color3.fromRGB(200,200,200)
    distLbl.Size=12; distLbl.Outline=true; distLbl.Center=true; distLbl.Font=Drawing.Fonts.UI
    espData[player]={lines=lines,box=box,hpBars={hpBack,hpFill},nametag=nametag,distLbl=distLbl}
end
local function updateESP(player)
    local char=player.Character; local hum=getHumanoid(char); local d=espData[player]
    if not char or not hum or not d then return end
    local alive=hum.Health>0
    -- ESP Drawing visible même en stream mode (Drawing API = pas capturé par OBS sur Roblox)
    local on=CONFIG.ESPEnabled and alive
    for i,conn in ipairs(SKELETON_CONNECTIONS) do
        local pA=getPart(char,conn[1]); local pB=getPart(char,conn[2]); local line=d.lines[i]
        if CONFIG.SkeletonEnabled and on and pA and pB then
            local sA,okA=worldToViewport(pA.Position); local sB,okB=worldToViewport(pB.Position)
            if okA and okB then line.From=sA; line.To=sB; line.Visible=true else line.Visible=false end
        else line.Visible=false end
    end
    local headPart=getPart(char,"Head"); local footPart=getPart(char,"LeftFoot") or getPart(char,"RightFoot")
    local hrp=getPart(char,"HumanoidRootPart"); local topSP,onTop,botSP,onBot
    if headPart and footPart then
        topSP,onTop=worldToViewport(headPart.Position+Vector3.new(0,0.6,0))
        botSP,onBot=worldToViewport(footPart.Position-Vector3.new(0,0.2,0))
    end
    if CONFIG.BoxEnabled and on and onTop and onBot then
        local h=math.abs(topSP.Y-botSP.Y); local w=h*0.55; local cx=topSP.X
        local tl=Vector2.new(cx-w/2,topSP.Y); local tr=Vector2.new(cx+w/2,topSP.Y)
        local bl=Vector2.new(cx-w/2,botSP.Y); local br=Vector2.new(cx+w/2,botSP.Y)
        local pts={tl,tr,tr,br,br,bl,bl,tl}
        for i2=1,4 do d.box[i2].From=pts[i2*2-1]; d.box[i2].To=pts[i2*2]; d.box[i2].Visible=true end
    else for _,l in ipairs(d.box) do l.Visible=false end end
    local hpBack=d.hpBars[1]; local hpFill=d.hpBars[2]
    if CONFIG.HealthBarEnabled and on and onTop and onBot then
        local h=math.abs(botSP.Y-topSP.Y); local cx=topSP.X; local xLeft=cx-h*0.55/2-6
        local pct=getHealthPercent(player); local barH=h*pct
        hpBack.From=Vector2.new(xLeft,topSP.Y); hpBack.To=Vector2.new(xLeft,botSP.Y); hpBack.Visible=true
        hpFill.From=Vector2.new(xLeft,botSP.Y); hpFill.To=Vector2.new(xLeft,botSP.Y-barH)
        hpFill.Color=lerpColor(CONFIG.HealthColorLow,CONFIG.HealthColorHigh,pct); hpFill.Visible=true
    else hpBack.Visible=false; hpFill.Visible=false end
    if CONFIG.NametagEnabled and on and headPart then
        local sp,okS=worldToViewport(headPart.Position+Vector3.new(0,2.4,0))
        if okS then d.nametag.Position=sp; d.nametag.Text=string.format("%s  [%d HP]",player.DisplayName,math.floor(hum.Health)); d.nametag.Visible=true
        else d.nametag.Visible=false end
    else d.nametag.Visible=false end
    if CONFIG.DistanceEnabled and on and hrp then
        local dist=getDistance(player); local sp,ok=worldToViewport(hrp.Position+Vector3.new(0,-2.0,0))
        if ok then d.distLbl.Position=sp; d.distLbl.Text=dist.." m"; d.distLbl.Visible=true
        else d.distLbl.Visible=false end
    else d.distLbl.Visible=false end
end

    end
end)
